cmake_minimum_required(VERSION 3.16)
project(etc2bridge)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)


# This pulls in etc2comp's OWN CMakeLists.txt, which already knows how to
# compile its core sources into a library target called `EtcLib`. That's the
# whole point of using CMake here instead of hand-listing .cpp files: let the
# library's own build script own its own file list.

add_subdirectory(etc2comp)


# EtcTool/ and are NOT part of EtcLib itself (they're normally only compiled
# into Etc2Comp's command-line tool). Since we need the .ktx file-writing
# code they contain, we compile them directly into OUR bridge library instead.

add_library(etc2bridge STATIC
    etc2comp_bridge.cpp
    etc2comp/EtcTool/EtcFile.cpp
    etc2comp/EtcTool/EtcFileHeader.cpp
)

target_include_directories(etc2bridge PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}      # for etc2comp_bridge.h and stb_image.h
    etc2comp/EtcLib/Etc
    etc2comp/EtcTool
)

# Links our bridge against Etc2Comp's own compiled core library.
target_link_libraries(etc2bridge PUBLIC EtcLib)
